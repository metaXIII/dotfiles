+switch-en() {
+    echo "=== Configuration du clavier en Anglais US (en_US / QWERTY) pour Fedora 44 ==="
+
+    # 1. Changement immédiat pour la session active (X11 et Wayland)
+    if command -v setxkbmap &> /dev/null; then
+        setxkbmap us
+    fi
+    # Support explicite pour GNOME Wayland
+    if command -v gsettings &> /dev/null && [[ "$XDG_CURRENT_DESKTOP" == *"GNOME"* ]]; then
+        gsettings set org.gnome.desktop.input-sources sources "[('xkb', 'us')]"
+    fi
+    # Support explicite pour KDE Plasma Wayland
+    if command -v kwriteconfig6 &> /dev/null && [[ "$XDG_CURRENT_DESKTOP" == *"KDE"* ]]; then
+        kwriteconfig6 --file kxkbrc --group Layout --key LayoutList "us"
+    fi
+    echo "[✓] Clavier de la session courante passé en en_US (QWERTY)."
+
+    # 2. Changement global et persistant du système
+    echo "-> Modification de la configuration système locale..."
+    sudo localectl set-x11-keymap us
+    sudo localectl set-keymap us
+    echo "[✓] Configuration locale système mise à jour en en_US."
+
+    # 3. Forcer Dracut à inclure la configuration du clavier pour le chiffrement
+    echo "-> Configuration de Dracut pour l'initramfs..."
+    if [ ! -f /etc/dracut.conf.d/keyboard.conf ]; then
+        echo 'i18n_install_all="yes"' | sudo tee /etc/dracut.conf.d/keyboard.conf > /dev/null
+        echo "[✓] Fichier de configuration Dracut créé."
+    else
+        echo "[i] Dracut est déjà configuré pour inclure les dispositions de clavier."
+    fi
+
+    # 4. Régénération de l'initramfs avec Dracut
+    echo "=== Régénération de l'image de démarrage (Initramfs) ==="
+    echo "Cette étape nécessite les privilèges root et prend quelques secondes..."
+    
+    sudo dracut -f --regenerate-all
+    
+    if [ $? -eq 0 ]; then
+        echo "[✓] Initramfs mis à jour avec succès pour tous les noyaux."
+        echo "---"
+        echo "Félicitations ! Votre clavier est configuré en en_US (QWERTY)."
+        echo "Au prochain redémarrage, l'écran de déverrouillage du disque sera en QWERTY."
+    else
+        echo "[X] Une erreur est survenue lors de la génération de l'initramfs."
+        return 1
+    fi
+}

+# Configuration Matérielle et Automatisations
+
+Ce dossier contient les scripts et configurations spécifiques à cette machine sous Fedora, notamment pour l'optimisation du stockage (partitionnement NVMe / SSD / HDD) et la gestion du cycle de vie des données.
+
+---
+
+## 📂 Structure du dossier
+
+- `scripts/`
+  - `archiver.sh` : Script Bash intelligent qui scanne les dossiers personnels (`Documents`, `Pictures`, `Music`, `Videos`) sur le NVMe et déplace automatiquement vers le HDD (`/mnt/HDD/Medias`) les fichiers inactifs depuis plus de 90 jours.
+- `system.d/`
+  - Sous-dossier `archive/` : Fichiers de configuration du service et du timer systemd utilisateur.
+
+---
+
+## 🛠️ Le Service d'Archivage Automatique
+
+### Objectif et Justification
+L'objectif est d'optimiser l'espace de stockage du NVMe principal (réservé au système, au code et aux notes) en déchargeant le contenu froid (médias, documents peu consultés) vers un HDD de stockage de masse.
+
+Le script utilise l'option `mv -n` (no-clobber) pour garantir qu'aucun fichier existant sur le HDD ne soit écrasé par mégarde.
+
+### Installation et Activation du Systemd Timer
+
+Pour que le système exécute cet archivage mensuellement de manière persistante (et rattrape la tâche au prochain démarrage si le PC était éteint à l'heure prévue) :
+
+1. **Créer les liens symboliques vers l'environnement utilisateur systemd :**
+
+```bash
+   mkdir -p ~/.config/systemd/user
+   ln -sf $(pwd)/system.d/archive/archiver.service ~/.config/systemd/user/archiver.service
+   ln -sf $(pwd)/system.d/archive/archiver.timer ~/.config/systemd/user/archiver.timer
+```
+
+# Consulter le dernier log d'archivage généré :
+cat /home/meta/Logs/archive/archive-*.log | tail -n 50
